const list = document.getElementById('quiz-list');
QUIZ.forEach((item, i) => {
  const card = document.createElement('div');
  card.className = 'quiz-card';
  card.innerHTML = '<h3>Q' + (i+1) + '. ' + item.q + '</h3>';
  const expl = document.createElement('div');
  expl.className = 'expl'; expl.textContent = item.expl;
  item.opts.forEach((opt, j) => {
    const b = document.createElement('button');
    b.className = 'opt'; b.textContent = opt;
    b.addEventListener('click', () => {
      card.querySelectorAll('.opt').forEach((o, k) => { o.classList.toggle('correct', k === item.ans); o.disabled = true; });
      if (j !== item.ans) b.classList.add('wrong');
      expl.classList.add('show');
    });
    card.appendChild(b);
  });
  card.appendChild(expl);
  list.appendChild(card);
});

<script>
(function(){
  var grid = document.getElementById('noteGrid');
  if (!grid) return;
  var rss = 'https://note.com/happy_ferret214/rss';
  fetch('https://api.allorigins.win/raw?url=' + encodeURIComponent(rss))
    .then(function(r){ if (!r.ok) throw new Error('bad'); return r.text(); })
    .then(function(xml){
      var doc = new DOMParser().parseFromString(xml, 'text/xml');
      var items = doc.querySelectorAll('item');
      if (!items.length) return;
      var html = '';
      for (var i = 0; i < Math.min(3, items.length); i++) {
        var t = items[i].querySelector('title').textContent;
        var l = items[i].querySelector('link').textContent;
        var d = new Date(items[i].querySelector('pubDate').textContent);
        var ds = d.getFullYear() + '年' + (d.getMonth() + 1) + '月' + d.getDate() + '日';
        html += '<a class="card" href="' + l + '" target="_blank" rel="noopener"><h3 style="font-size:17px">' +
          t.replace(/&/g,'&amp;').replace(/</g,'&lt;') + '</h3><p class="small-note">' + ds + ' ・ note</p></a>';
      }
      if (html) grid.innerHTML = html;
    })
    .catch(function(){});
})();
</script>
